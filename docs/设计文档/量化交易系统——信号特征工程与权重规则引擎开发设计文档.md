一、项目背景与目标
1.1 背景描述
通过对 Smooth 策略 在 ETH-USDT-SWAP 5分钟K线 上的历史信号数据进行系统性分析（实际勘测 6,209 笔有效交替），识别了以下 7 条核心规律：

序号	规律	数据支撑	应用方向
1	信号越密集，空间越小	间隔 < 1 小时，平均空间 < 0.3%	低波过滤
2	间隔越长，空间越大	>24h 信号空间是 <1h 的 3~5 倍	时间权重
3	连续同向有累积效应	连续 3 次同向，平均累积空间 3.8%	趋势加仓
4	趋势第 2~3 笔质量最高	第 3 笔平均空间 +2.10%，第 4 笔后衰竭	趋势阶段识别
5	极端空间后必反转	>5% 信号，后续 72%+ 概率反向	极端保护 + 反向进攻
6	状态有强惯性	震荡延续 73%，趋势延续 68%，低波延续 81%	市场状态识别
7	方向偏置显著	多头净贡献 +497.23%，空头 -312.57%	方向偏置因子
方向偏置数据说明（基于 6,209 笔实际数据统计）：

多头信号（LONG→SHORT）：3,452 笔，胜率 55.7%，净贡献 +497.23%

空头信号（SHORT→LONG）：2,757 笔，胜率 47.6%，净贡献 -312.57%

多空净贡献差值：809.80%

当前系统已具备：

✅ 信号生成与 AI 过滤能力

✅ 配置化权重规则引擎（WeightRuleEngine + signal_service_config.weight_rules_json）

✅ SPI 指标注册机制（IndicatorProvider + BuiltInIndicatorProvider）

✅ 多类型指标值（NUMERIC / STRING / BOOLEAN）

但缺少 信号特征工程层（L2 统计特征），导致上述规律未被充分利用。

FROM technical_signal ts

p.strategy_name,
p.symbol,
p.timeframe,
p.entry_time, p.entry_price, p.entry_direction, p.entry_signal_id,
p.exit_time, p.exit_price, p.exit_direction, p.exit_signal_id,

5.3 滑动窗口数据结构（SlidingWindow）
java
package com.chain.ai.trade.engine.signal.feature;

import java.util.*;

/**
* 滑动窗口 —— 核心数据结构
* 模块归属：ai-signal
* 职责：
* 1. 维护最近 20 笔交替记录的 space 和 direction
* 2. 提供基础统计值：avgSpace, cumRatio, directionSeq 等
* 3. 提供动态分位数计算
* 4. 实盘和回测使用完全相同的逻辑
     */
     public class SlidingWindow {
     private final Queue<Double> spaceQueue = new LinkedList<>();
     private final Queue<Double> absSpaceQueue = new LinkedList<>();
     private final Queue<String> directionQueue = new LinkedList<>();
     private final Queue<Long> timeQueue = new LinkedList<>();
     private final int maxSize;

private double cumSum = 0.0;
private double absSum = 0.0;
private long lastSignalTime = 0L;
private double latestSpace = 0.0;
private String lastDirection = null;
private int seq = 0;

public SlidingWindow(int maxSize) {
this.maxSize = maxSize;
}

/**
    * 添加一笔交替记录，更新滑动窗口
    * ★ 实盘和回测使用完全相同的逻辑
      */
      public void add(SignalAlternateLog record) {
      double space = record.getSpacePct();
      double absSpace = Math.abs(space);
      String direction = record.getEntryDirection();
      long entryTime = record.getEntryTime();

      // 入队
      spaceQueue.add(space);
      absSpaceQueue.add(absSpace);
      directionQueue.add(direction);
      timeQueue.add(entryTime);
      cumSum += space;
      absSum += absSpace;
      lastSignalTime = entryTime;
      latestSpace = space;

      // 更新方向序列
      if (direction.equals(lastDirection)) {
      seq++;
      } else {
      seq = 1;
      }
      lastDirection = direction;

      // 超出窗口则移除最旧值
      if (spaceQueue.size() > maxSize) {
      spaceQueue.poll();
      absSpaceQueue.poll();
      directionQueue.poll();
      timeQueue.poll();
      recalcFromQueue();
      }
      }

// ========== 基础统计 ==========

public double getAvgAbsSpace() {
return absSum / Math.max(1, spaceQueue.size());
}

public double getCumRatio() {
return absSum == 0 ? 0 : cumSum / absSum;
}

public int getDirectionSeq() {
return seq;
}

public long getLastSignalTime() {
return lastSignalTime;
}

public long getFirstSignalTime() {
return timeQueue.isEmpty() ? 0 : timeQueue.peek();
}

public double getLatestSpace() {
return latestSpace;
}

public String getLastDirection() {
return lastDirection;
}

public int getSize() {
return spaceQueue.size();
}

// ========== 动态分位数计算 ==========

public double getAbsSpacePercentile(double percentile) {
if (absSpaceQueue.isEmpty()) return 0.0;
List<Double> sorted = new ArrayList<>(absSpaceQueue);
Collections.sort(sorted);
int index = (int) Math.ceil(percentile * sorted.size()) - 1;
index = Math.max(0, Math.min(index, sorted.size() - 1));
return sorted.get(index);
}

public double getSpacePercentile(double percentile) {
if (spaceQueue.isEmpty()) return 0.0;
List<Double> sorted = new ArrayList<>(spaceQueue);
Collections.sort(sorted);
int index = (int) Math.ceil(percentile * sorted.size()) - 1;
index = Math.max(0, Math.min(index, sorted.size() - 1));
return sorted.get(index);
}

public FeatureStatistics getFullStatistics() {
FeatureStatistics stats = new FeatureStatistics();
stats.setAvgSpace(getAvgAbsSpace());
stats.setCumRatio(getCumRatio());
stats.setDirectionSeq(getDirectionSeq());
stats.setFirstSignalTime(getFirstSignalTime());
stats.setLastSignalTime(getLastSignalTime());
stats.setLatestSpace(getLatestSpace());
stats.setLastDirection(getLastDirection());
stats.setWindowSize(getSize());
stats.setPercentile_20(getAbsSpacePercentile(0.20));
stats.setPercentile_40(getAbsSpacePercentile(0.40));
stats.setPercentile_70(getAbsSpacePercentile(0.70));
stats.setPercentile_85(getAbsSpacePercentile(0.85));
stats.setPercentile_95(getAbsSpacePercentile(0.95));
stats.setCumRatioPercentile_40(getSpacePercentile(0.40));
stats.setCumRatioPercentile_60(getSpacePercentile(0.60));
return stats;
}

// ========== 私有方法 ==========

private void recalcFromQueue() {
cumSum = spaceQueue.stream().mapToDouble(Double::doubleValue).sum();
absSum = absSpaceQueue.stream().mapToDouble(Double::doubleValue).sum();
seq = recalcDirectionSeq();
}

private int recalcDirectionSeq() {
if (directionQueue.isEmpty()) return 0;
int count = 1;
String[] dirs = directionQueue.toArray(new String[0]);
for (int i = dirs.length - 2; i >= 0; i--) {
if (dirs[i].equals(dirs[i + 1])) {
count++;
} else {
break;
}
}
return count;
}
}

private static boolean evaluateNumericCondition(Double actual, OperatorType op,
String expected, WeightRuleContext ctx) {
if (actual == null || expected == null) return false;

    double expectedVal;
    try {
        expectedVal = Double.parseDouble(expected);
    } catch (NumberFormatException e) {
        return false;
    }
    
    switch (op) {
        // 原有运算符保持不变...
        case GT: return actual > expectedVal;
        case GTE: return actual >= expectedVal;
        case LT: return actual < expectedVal;
        case LTE: return actual <= expectedVal;
        // ... 其他原有运算符 ...
        
        // ===== 新增：abs_space 分位数比较 =====
        case LT_ABS_PERCENTILE: {
            double p = getAbsSpacePercentile(ctx, expectedVal);
            return actual < p;
        }
        case GT_ABS_PERCENTILE: {
            double p = getAbsSpacePercentile(ctx, expectedVal);
            return actual > p;
        }
        case LTE_ABS_PERCENTILE: {
            double p = getAbsSpacePercentile(ctx, expectedVal);
            return actual <= p;
        }
        case GTE_ABS_PERCENTILE: {
            double p = getAbsSpacePercentile(ctx, expectedVal);
            return actual >= p;
        }
        
        // ===== 新增：cumRatio 分位数比较 =====
        case LT_RATIO_PERCENTILE: {
            double p = getCumRatioPercentile(ctx, expectedVal);
            return actual < p;
        }
        case GT_RATIO_PERCENTILE: {
            double p = getCumRatioPercentile(ctx, expectedVal);
            return actual > p;
        }
        case LTE_RATIO_PERCENTILE: {
            double p = getCumRatioPercentile(ctx, expectedVal);
            return actual <= p;
        }
        case GTE_RATIO_PERCENTILE: {
            double p = getCumRatioPercentile(ctx, expectedVal);
            return actual >= p;
        }
        
        // ===== 新增：方向比较 =====
        case EQ_CURRENT_SAME: {
            return ctx.getLastDirection() != null && 
                   ctx.getCurrentDirection() != null &&
                   ctx.getLastDirection().equals(ctx.getCurrentDirection());
        }
        case EQ_CURRENT_OPPOSITE: {
            return ctx.getLastDirection() != null && 
                   ctx.getCurrentDirection() != null &&
                   !ctx.getLastDirection().equals(ctx.getCurrentDirection());
        }
        default: return false;
    }

        if (avgSpace < 0.4 && avgSpace > 0) {
            log.info("低波过滤拦截(降级模式): strategy={}, avgSpace={}%", strategyName, avgSpace);
            technicalSignalService.markRejected(result.getSignalId(), "低波过滤(降级阈值0.4%)");
            return;
        }
    }
    
    // 3. AI过滤 + 保存信号
    TechnicalSignal signal = applyAiFilter(result, ctx, latestKline.getId());
    pushSignalToRedis(result, ctx, signal);
    redisCache.setCacheObject(cacheKey, System.currentTimeMillis(), expireMinutes, TimeUnit.MINUTES);
}
